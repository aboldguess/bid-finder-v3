            // Block Authority Control Panel access for users with no active subscription
            (function() {
                var noActiveSubscription = 'false';

                if (noActiveSubscription === 'true') {

                    // Function to block links
                    function blockControlPanelLinks() {
                        // Find all links that point to Authority Control Panel - use multiple selectors
                        var authorityLinks = document.querySelectorAll('a[href*="Authority_ControlPanel"], a[href*="authority_controlpanel"], a[href*="Authority/Authority_ControlPanel"]');

                        // Find all links that point to Member/Supplier Control Panel
                        var memberLinks = document.querySelectorAll('a[href*="Member_ControlPanel"], a[href*="member_controlpanel"], a[href*="Membership/Member_ControlPanel"]');

                        // Block authority links
                        if (authorityLinks.length === 0) {
                            // Try again with all links
                            var allLinks = document.querySelectorAll('a');

                            allLinks.forEach(function (link) {
                                var href = link.getAttribute('href') || '';
                                if (href.toLowerCase().indexOf('authority_controlpanel') > -1) {
                                    blockLink(link, 'Authority');
                                }
                            });
                        } else {
                            authorityLinks.forEach(function (link) {
                                blockLink(link, 'Authority');
                            });
                        }

                        // Block member/supplier links
                        if (memberLinks.length === 0) {
                            // Try again with all links
                            var allLinks = document.querySelectorAll('a');

                            allLinks.forEach(function (link) {
                                var href = link.getAttribute('href') || '';
                                if (href.toLowerCase().indexOf('member_controlpanel') > -1) {
                                    blockLink(link, 'Member');
                                }
                            });
                        } else {
                            memberLinks.forEach(function (link) {
                                blockLink(link, 'Member');
                            });
                        }
                    }

                    // Function to block a single link
                    function blockLink(link, panelType) {

                        // Remove any existing click handlers
                        var newLink = link.cloneNode(true);

                        // Add new click handler
                        newLink.addEventListener('click', function (e) {
                            e.preventDefault();
                            e.stopPropagation();
                            e.stopImmediatePropagation();

                            var message = panelType === 'Authority'
                                ? 'You do not have an active subscription. Please renew your subscription to access the Authority Control Panel.'
                                : 'You do not have an active subscription. Please renew your subscription to access the Member Control Panel.';

                            alert(message);

                            // Mark that we've shown the alert, so it doesn't show again on subscription page
                            sessionStorage.setItem('subscriptionAlertShown', 'true');

                            // Redirect to subscription page
                            window.location.href = '/Membership/Member_Settings.aspx?page=subs';

                            return false;
                        }, true); // Use capture phase

                        // Add visual indicator
                        newLink.style.opacity = '0.6';
                        newLink.style.cursor = 'not-allowed';
                        newLink.title = 'Active subscription required';

                        // Replace the old link with new one
                        if (link.parentNode) {
                            link.parentNode.replaceChild(newLink, link);
                        }
                    }

                    // Run when DOM is ready
                    if (document.readyState === 'loading') {
                        document.addEventListener('DOMContentLoaded', blockControlPanelLinks);
                    } else {
                        // DOM already loaded
                        blockControlPanelLinks();
                    }

                    // Also run after a short delay to catch dynamically loaded content
                    setTimeout(blockControlPanelLinks, 1000);
                    setTimeout(blockControlPanelLinks, 2000);
                }
            })();
    </script>
