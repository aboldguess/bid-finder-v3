	function esc(s) {
		return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
			return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
		});
	}

	function nl2br(s) {
		return String(s).replace(/\r\n|\r|\n/g, '<br>');
	}

	function eq(a, b) {
		a = String(a == null ? '' : a);
		b = String(b == null ? '' : b);
		if (a === b) {
			return true;
		}
		return a !== '' && b !== '' && /^\d+$/.test(a) && /^\d+$/.test(b) && parseInt(a, 10) === parseInt(b, 10);
	}

	function toYmd(v) {
		if (!v) {
			return '';
		}
		v = String(v);
		var m = v.match(/(\d{4})\D+(\d{1,2})\D+(\d{1,2})/);
		if (m) {
			return m[1] + ('0' + m[2]).slice(-2) + ('0' + m[3]).slice(-2);
		}
		m = v.match(/^(\d{8})$/);
		return m ? m[1] : '';
	}

	function todayYmd() {
		var d = new Date();
		return '' + d.getFullYear() + ('0' + (d.getMonth() + 1)).slice(-2) + ('0' + d.getDate()).slice(-2);
	}

	var MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
	function fmtDate(ymd) {
		if (!ymd || ymd.length !== 8) {
			return '';
		}
		var y = ymd.substr(0, 4), m = ymd.substr(4, 2), d = ymd.substr(6, 2);
		if (C.lang === 'en') {
			return MONTHS[parseInt(m, 10) - 1] + ' ' + d + ', ' + y;
		}
		return y + '年' + m + '月' + d + '日';
	}

	function strimwidth(s, width) {
		var w = 0, out = '';
		for (var i = 0; i < s.length; i++) {
			var cw = s.charCodeAt(i) > 0xFF ? 2 : 1;
			if (w + cw > width - 3) {
				var rest = 0;
				for (var j = i; j < s.length; j++) {
					rest += s.charCodeAt(j) > 0xFF ? 2 : 1;
				}
				if (w + rest <= width) {
					return s;
				}
				return out + '...';
			}
			w += cw;
			out += s.charAt(i);
		}
		return out;
	}

	function parseQuery(qs) {
		var o = {};
		qs = String(qs || '').replace(/^\?/, '');
		if (!qs) {
			return o;
		}
		var parts = qs.split('&');
		for (var i = 0; i < parts.length; i++) {
			if (!parts[i]) {
				continue;
			}
			var kv = parts[i].split('=');
			var k, v;
			try {
				k = decodeURIComponent(kv[0].replace(/\+/g, ' '));
				v = decodeURIComponent((kv.slice(1).join('=') || '').replace(/\+/g, ' '));
			} catch (e) {
				continue;
			}
			o[k] = v;
		}
		return o;
	}

	function condFromParams(q) {

	function hasCond(q) {

	}

	function formParams() {

	function condQuery(c) {

		for (var i = 0; i < KEYS.length; i++) {
			var k = KEYS[i];
			if (c[k] !== undefined && c[k] !== '') {
				q.push(encodeURIComponent(C.prefix + k) + '=' + encodeURIComponent(c[k]));
			}
		}
		return q.join('&');
	}

	function pageUrl(p) {
		var q = condQuery(cond);
		return location.pathname + '?' + (q ? q + '&' : '') + '_page=' + p;
	}

	function compile(c) {
		var r = {};
		r.from = toYmd(c.from);
		r.to = toYmd(c.to);
		r.entity = c.entity || '';
		r.words = [];
		if (c.keyword) {
			var ws = String(c.keyword).split(/[\s　]+/);
			for (var i = 0; i < ws.length; i++) {
				if (ws[i] !== '') {
					r.words.push(ws[i].toLowerCase());
				}
			}
		}
		r.cats = [];
		for (var n = 1; n <= 3; n++) {
			if (c['classification' + n]) {
				r.cats.push(c['classification' + n]);
			}
		}
		r.dlFrom = '';
		r.dlTo = '';
		if (c.deadline == 2) {
			r.dlFrom = toYmd(c.deadline_from);
			r.dlTo = toYmd(c.deadline_to);
		} else if (c.deadline == 1) {
			r.dlFrom = todayYmd();
		}
		r.type = isCentral ? (c.type || '') : '';
		r.area = isCentral ? (c.area || '') : '';
		return r;
	}

	function match(row, r) {
		var d = row[X.date];
		if (r.from && (!d || d < r.from)) { return false; }
		if (r.to && (!d || d > r.to)) { return false; }
		if (r.entity && !eq(row[X.agency], r.entity)) { return false; }
		if (r.words.length) {
			var t = String(row[X.title]).toLowerCase(), hit = false;
			for (var i = 0; i < r.words.length; i++) {
				if (t.indexOf(r.words[i]) !== -1) { hit = true; break; }
			}
			if (!hit) { return false; }
		}
		if (r.cats.length) {
			var rc = row[X.cats] ? String(row[X.cats]).split('|') : [], found = false;
			for (var a = 0; a < rc.length && !found; a++) {
				for (var b = 0; b < r.cats.length; b++) {
					if (eq(rc[a], r.cats[b])) { found = true; break; }
				}
			}
			if (!found) { return false; }
		}
		var dl = row[X.deadline];
		if (r.dlFrom && (!dl || dl < r.dlFrom)) { return false; }
		if (r.dlTo && (!dl || dl > r.dlTo)) { return false; }
		if (r.type && !eq(row[X.type], r.type)) { return false; }
		if (r.area && !eq(row[X.location], r.area)) { return false; }
		return true;
	}

	}

	function master(name, code) {

		for (var k in m) {
			if (m.hasOwnProperty(k) && eq(k, code)) {
				return m[k];
			}
		}
		return '';
	}

	function detailHref(row) {

	function renderList() {
		var total = filtered.length;
		var perPage = C.perPage;
		var lastPage = Math.ceil(total / perPage);
		if (page > lastPage) { page = Math.max(1, lastPage); }

	function renderPagination(lastPage) {
		var $ul = $('#xiProcPagination');
		var $wrap = $ul.closest('.elem_pagination');
		if (lastPage <= 0) {
			$wrap.hide();
			$ul.empty();

		function li(p, text, attr) {
			h += '<li' + (attr || '') + '><a href="' + esc(pageUrl(p)) + '" data-page="' + p + '">' + esc(text) + '</a></li>';
		}
		if (cur > 0) { h += '<li id="prevPage"><a class="pageprev" href="' + esc(pageUrl(page - 1)) + '" data-page="' + (page - 1) + '">' + esc(L.prev) + '</a></li>'; }
		if (cur >= 5) { li(1, '1', ' id="firstPage"'); }
		if (cur >= 6) { h += '<li id="morePrev"><a>...</a></li>'; }
		for (var i = 4; i >= 1; i--) {
			if (cur - i >= 0) { li(cur - i + 1, cur - i + 1); }
		}
		h += '<li id="pageCurrent" class="current"><a>' + page + '</a></li>';
		for (var j = 1; j <= 4; j++) {
			if (cur + j < lastPage) { li(cur + j + 1, cur + j + 1); }
		}
		if (cur <= lastPage - 7) { h += '<li id="moreNext"><a>...</a></li>'; }
		if (cur <= lastPage - 6) { li(lastPage, lastPage, ' id="lastPage"'); }
		if (cur < lastPage - 1) { h += '<li id="nextPage"><a class="pagenext" href="' + esc(pageUrl(page + 1)) + '" data-page="' + (page + 1) + '">' + esc(L.next) + '</a></li>'; }
		$ul.html(h);
		$wrap.show();
	}

	function scrollTop() {
		var $t = $('#mainArea');
		if ($t.length) {
			$('html,body').animate({ scrollTop: $t.offset().top }, 'fast');
		}
	}

	function pushUrl() {
		var url = pageUrl(page);
		if (window.history && window.history.pushState && url !== location.pathname + location.search) {
			window.history.pushState({ xiProc: 1 }, '', url);
		}
	}

	function message(text) {
		$('#xiProcResults').html('<tr class="appendResult"><td colspan="4">' + esc(text) + '</td></tr>');
		$('#xiProcPagination').closest('.elem_pagination').hide();
	}

	function loadFromLocation() {
		var q = parseQuery(location.search);
		cond = hasCond(q) ? condFromParams(q) : condFromParams(formParams());
		page = Math.max(1, parseInt(q._page, 10) || 1);
	}

	function run() {
		search();
		renderCondition();
		renderList();
	}

	function bindEvents() {
		$('#xiProcPagination').on('click', 'a[data-page]', function () {
			page = parseInt($(this).data('page'), 10) || 1;
			renderList();
			pushUrl();
			scrollTop();
			return false;
		});
		$('#xiProcResults').on('click', 'tr[data-href]', function () {
			location.href = $(this).data('href');
			return false;
		});
		var doSearch = function () {
			if (!json) {
				return false;
			}
			cond = condFromParams(formParams());
			page = 1;
			run();
			pushUrl();
			$('#block_search').slideUp();
			$('#block_condition').slideDown();
			return false;
		};
		$(C.searchBtn).off('click').on('click', doSearch);
		$('#' + C.formId).off('submit').on('submit', doSearch);
		$('#open_search').off('click').on('click', function () {
			$('#block_search').slideToggle();
			$('#block_condition').slideToggle();
			return false;
		});
		$(window).on('popstate', function () {
			if (!json) {
				return;
			}
			loadFromLocation();
			run();
		});
	}

	$(function () {
		bindEvents();
		loadFromLocation();
		if (!C.jsonUrl) {
			message(L.error);
			return;
		}
		message(L.loading);
		$.ajax({
			url: C.jsonUrl,
			dataType: 'json',
			cache: true,
			success: function (data) {
				if (!data || !data.rows) {
					message(L.error);
					return;
				}
				json = data;
				run();
			},
			error: function () {
				message(L.error);
			}
		});
	});
})(jQuery);

				GSSA.set("amount", data.amount);
				GSSA.send();

				GSSA.set('page', data.page);

				GSSA.send();
			}
			else if (data.mode == 'other') {

				GSSA.send();
